# libthv8.so: the app's JNI bridge linked against a small static V8 (v8-build branch's CI: libv8_monolith.a + include)
cmake_minimum_required(VERSION 3.22)
project(thv8 CXX)
set(CMAKE_CXX_STANDARD 20)
set(V8 ${CMAKE_CURRENT_SOURCE_DIR}/../../../v8)
add_library(thv8 SHARED thv8.cc)
target_include_directories(thv8 PRIVATE ${V8}/include)
target_compile_definitions(thv8 PRIVATE V8_COMPRESS_POINTERS V8_31BIT_SMIS_ON_64BIT_ARCH V8_ENABLE_SANDBOX)
target_compile_options(thv8 PRIVATE -Os -fvisibility=hidden -ffunction-sections -fdata-sections)
target_link_libraries(thv8 ${V8}/lib/libv8_monolith.a log)
target_link_options(thv8 PRIVATE -Wl,--gc-sections -Wl,--icf=all -Wl,--exclude-libs,ALL -s)
